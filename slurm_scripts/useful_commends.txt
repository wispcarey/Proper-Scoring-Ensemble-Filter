# --- tmux Session Management ---

# Create a new session with a specific name
tmux new -s jupyter_session

# Detach from the current session (leave it running in background)
# Shortcut: Ctrl+b then d
tmux detach

# List all active sessions
tmux ls

# Re-attach to a specific session
tmux attach -t jupyter_session

# Kill a specific session
tmux kill-session -t jupyter_session
